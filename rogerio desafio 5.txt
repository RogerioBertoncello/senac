import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Digite seu nome: ");
        String nome = scanner.nextLine();

        System.out.print("Digite sua idade: ");
        int idade = scanner.nextInt();
        scanner.nextLine();

        if (idade < 18) {
            System.out.println("Desculpe, " + nome + ". Apenas maiores de 18 anos podem reservar ingressos.");
            scanner.close();
            return;
        }

        System.out.print("Você é estudante de Python? (Sim/Não): ");
        String respostaEstudante = scanner.nextLine().trim().toLowerCase();

        boolean isEstudantePython = respostaEstudante.startsWith("s");

        System.out.println("\nEscolha o tipo de ingresso:");
        System.out.println("1 - Entrada Padrão (R$ 20,00)");
        System.out.println("2 - Entrada VIP (R$ 50,00)");
        System.out.print("Digite o número da opção desejada: ");
        int opcao = scanner.nextInt();

        double precoBase;

        if (opcao == 1) {
            precoBase = 20.0;
        } else if (opcao == 2) {
            precoBase = 50.0;
        } else {
            System.out.println("Opção inválida. Reserva cancelada.");
            scanner.close();
            return;
        }

        double precoFinal = precoBase;
        if (isEstudantePython) {
            precoFinal = precoBase * 0.5;
            System.out.println("\nDesconto de 50% aplicado para estudante de Python!");
        }

        System.out.println("\n--- Reserva Confirmada ---");
        System.out.println("Cliente: " + nome);
        System.out.println("Idade: " + idade + " anos");
        System.out.printf("Valor total a pagar: R$ %.2f\n", precoFinal);

        scanner.close();
    }
}
