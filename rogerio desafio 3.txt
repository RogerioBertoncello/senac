package org.example;

import java.util.Scanner;

public class Main {
    public static void main(String[] args) {

        int acerto = 0;

        try (Scanner scanner = new Scanner(System.in)) {

            System.out.println("Quiz da Computação");

            System.out.println("Quem foi Alan Turing?\na) Integrante de uma banda\nb) Pai da Ciência da Computação\nc) Figurante de uma série");
            String resposta1 = scanner.nextLine().trim();
            if (resposta1.equalsIgnoreCase("b")) {
                acerto++;
            }

            System.out.println("Quem foi Ada Lovelace?\na) Jogadora de futebol\nb) Primeira programadora mulher\nc) Matemática e escritora");
            String resposta2 = scanner.nextLine().trim();
            if (resposta2.equalsIgnoreCase("b")) {
                acerto++;
            }

            System.out.println("Quem foi Steve Jobs?\na) Vendedor de mandolate\nb) Inventor, empresário e magnata\nc) Policial");
            String resposta3 = scanner.nextLine().trim();
            if (resposta3.equalsIgnoreCase("b")) {
                acerto++;
            }

            System.out.println("Você acertou " + acerto + "perguntas!");
        }
    }
}
